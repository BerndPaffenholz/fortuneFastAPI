#+TITLE: REST-Service für fortune with FastAPI

* Motivation

Nachdem es mir nicht gelungen ist, einen verläßlichen Service für das
Bereitstellen von "fortunes" zu finden, habe ich einen selbst erstellt.

Es gibt das Programm [[https://linux.die.net/man/6/fortune][=fortune(6)=]], aber das ist nicht immer
installiert. Außerdem wollte ich einen Service programmieren.

* Lösung

Die Datenbasis sollte die gleiche wie für das Programm =fortune=
sein. Die aktuelle Quelle ist das [[https://github.com/shlomif/fortune-mod][Projekt fortune-mod]].  Ich verwende
aber einen [[https://github.com/BerndPaffenholz/fortune-mod.git][Klon]], den ich kontrollieren kann, damit ich Ergänzungen
oder Streichungen vornehmen kann.

Die Textdateien mit den Zitaten werden in eine Datenbank geladen. Der
Service liest aus der Datenbank.

** Anwendung und REST-Schnitstelle

import logging
log = logging.getLogger("fortunes")

from pymongo import MongoClient
mongodb = MongoClient("mongodb://mongo:27017/",username="root",password="example")
databasename = "fortune"

def get_citesdb():
    global mongodb, databasename
    try:
        return mongodb[databasename].sayings
    except Exception as e:
        print(e)
        raise e

def get_confdb():
    global mongodb, databasename
    try:
        return mongodb[databasename].config
    except Exception as e:
        print(e)
        raise e

def close_db(e=None):
    global mongodb, databasename
    if mongodb is not None:
        pass

def find_one (cat=None):
    citesdb = get_citesdb()
    if cat == None:
        query = []
        keyerror_msg = "Nothing found."
        error_msg = "Problem with database."
    else:
        query = [ { "$match": { "category": cat } } ]
        keyerror_msg = f"Category '{cat}' not found."
        error_msg = f"Problem with category '{cat}'."
    query += [ { "$sample": { "size": 1 } }]
    try:
        sample = citesdb.aggregate(query).next()
        return { k:sample[k] for k in ("text", "category") }
    except StopIteration :
        raise KeyError(keyerror_msg)
    except Exception as e:
        log.error(e)
        raise LookupError(error_msg)

def list_categories ():
    citesdb = get_citesdb()
    counts = citesdb.aggregate([{ "$group": { "_id": "$category", "count": { "$count": {} } } }])
    dict = {}
    for c in counts:
        dict[c["_id"]] = c["count"]
    return dict

def init_app(app):
    # app.teardown_appcontext(close_db)
    global mongodb, databasename
    if not (databasename in  mongodb.list_database_names()):
        log.info("no database")
        mongodb[databasename].config.insert_one({"name": 'git-commit-id', "value": None})
        return False
    else:
        return True

def add_saying_with_category (saying, category):
    citesdb = get_citesdb()
    citesdb.insert_one({"category": category, "text": saying })

def remove_category (category):
    citesdb = get_citesdb()
    citesdb.delete_many({"category": category })

def read_category_file_add_to_collection (file):
    count = 0
    saying = ""
    category = file.name

    remove_category(category)

    with file.open("r") as fd:
        for l in fd:
            if l == "%\n":
                if saying != "":
                    count += 1
                    add_saying_with_category (saying, category)
                    saying = ""
            else:
                saying += l

    return {f"{category}": count}

from pathlib import Path

def list_files_for_cites ():
    source = Path('/fortune-mod/fortune-mod/datfiles')
    lof = [str(f) for f in source.iterdir() if f.is_file() and f.suffix == ""]
    return lof

from subprocess import check_output, call

def get_current_git_commit_id ():
    call(['git', 'pull', '--ff-only'],cwd="/fortune-mod")
    return check_output(['git', 'log', '-1', '--format=%H'], shell=False, text=True,cwd="/fortune-mod")

def get_db_git_commit_id ():
    config = get_confdb()
    commit_value = config.find_one({"name":"git-commit-id"})
    if commit_value == None:
        return ""
    else:
        return commit_value["value"] or ""

def update_git_commit_id ():
    config = get_confdb()
    git_id = get_current_git_commit_id()
    config.update_one({"name":"git-commit-id"}, {"$set":{"value":git_id}}, upsert=True)

#+begin_src python :results output
from celery import Celery, chain, group
from celery.schedules import crontab
import app.cites as cites
from celery.utils.log import get_task_logger

from pathlib import Path

#import logging
#logger = logging.getLogger("fortunes")

app = Celery('tasks', backend='rpc://guest@rabbitmq', broker='pyamqp://guest@rabbitmq//')
app.conf.timezone = 'Europe/Berlin'
logger = get_task_logger("fortunes")

@app.on_after_configure.connect
def setup_periodic_tasks(sender, **kwargs):
    # Executes every morning at 0500
    sender.add_periodic_task(
        crontab(minute=0,hour=5),
        run_update_db.s()
    )

@app.task
def run_update_db():
    logger.info("Run the task 'run_update_db'.")
    if cites.get_current_git_commit_id() != cites.get_db_git_commit_id():
        print("Mid in run_update_db")
        res = chain(list_files.s(), map_in_group.s(), compare_categories.s(), update_gitid.si())
        return res()
    return False

@app.task
def list_files():
    logger.info("list files")
    return cites.list_files_for_cites()

@app.task
def map_in_group (filelist):
    return group([load_one_file(f) for f in filelist])
    
@app.task(ignore_result=False)
def load_one_file(filename):
    return cites.read_category_file_add_to_collection(Path(filename))

@app.task(ignore_result=False)
def compare_categories (results):
    from functools import reduce
    import operator
    new_loaded_categories = reduce(operator.or_,results["kwargs"]["tasks"])
    current_categories = cites.list_categories()
    to_be_dropped = filter(lambda cat: cat not in new_loaded_categories, current_categories.keys())
    for cat in to_be_dropped:
        logger.info("dropping %s", cat)
        cites.remove_category(cat)
    return True

@app.task
def update_gitid ():
    logger.info("Update git id")
    cites.update_git_commit_id()
    return True

wird das durch den Updatemechanismus nicht berücksichtigt. Die Daten

nur ein Löschen der Mongo-DB durch Neustart des Containers mit
gelöschten Volumes helfen.
